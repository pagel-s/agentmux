func TestPilotWorkerSecretExtraction(t *testing.T) {
	t.Setenv("PILOT_SELECTED_API", "ambient-selected-provider-secret")
	req := harness.Request{APIKeyEnv: "PILOT_SELECTED_API", Env: map[string]string{
		"CUSTOM_TOKEN":   "overlapping-secret-long",
		"CUSTOM_KEY":     "duplicate-secret",
		"PASSWORD_VALUE": "password-with-characters-<&>",
		"SECRET_HINT":    "secret-hint-value",
		"EMPTY_KEY":      "",
		"ORDINARY":       "ordinary-non-secret",
	}}
	got := workerSecrets(req, "broker-owner-token", "duplicate-secret", "", "overlapping-secret")
	wanted := []string{"broker-owner-token", "duplicate-secret", "overlapping-secret", "overlapping-secret-long", "ambient-selected-provider-secret", "password-with-characters-<&>", "secret-hint-value"}
	seen := map[string]bool{}
	for i, value := range got {
		if value == "" || seen[value] || i > 0 && len(got[i-1]) < len(value) {
			t.Fatalf("secrets must be nonempty, unique and longest first: %q", got)
		}
		seen[value] = true
	}
	if len(seen) != len(wanted) {
		t.Fatalf("wrong secret set: %q", got)
	}
	for _, value := range wanted {
		if !seen[value] {
			t.Fatalf("missing selected secret %q", value)
		}
	}
	if seen[req.Env["ORDINARY"]] {
		t.Fatal("ordinary environment value became a credential")
	}
	before := map[string]string{}
	for key, value := range req.Env {
		before[key] = value
	}
	_ = workerSecrets(req, "broker-owner-token", "duplicate-secret")
	if !reflect.DeepEqual(req.Env, before) {
		t.Fatal("secret inspection mutated the harness environment")
	}
	// An explicit request value, including an empty value, takes precedence over
	// ambient credentials; collection must not accidentally broaden authority.
	req.Env["PILOT_SELECTED_API"] = "request-selected-provider-secret"
	selected := workerSecrets(req, "")
	if !pilotContains(selected, "request-selected-provider-secret") || pilotContains(selected, "ambient-selected-provider-secret") {
		t.Fatalf("selected API request value lost precedence: %q", selected)
	}
	req.Env["PILOT_SELECTED_API"] = ""
	if pilotContains(workerSecrets(req, ""), "ambient-selected-provider-secret") {
		t.Fatal("explicit empty selected API value fell back to ambient authority")
	}
}

func TestPilotWorkerResultUsesSharedSecretCollector(t *testing.T) {
	packages, err := parser.ParseDir(token.NewFileSet(), ".", func(info os.FileInfo) bool { return !strings.HasSuffix(info.Name(), "_test.go") }, 0)
	if err != nil {
		t.Fatal(err)
	}
	found := false
	for _, pkg := range packages {
		for _, file := range pkg.Files {
			for _, declaration := range file.Decls {
				function, ok := declaration.(*ast.FuncDecl)
				if !ok || function.Name.Name != "workerResult" {
					continue
				}
				ast.Inspect(function.Body, func(node ast.Node) bool {
					call, ok := node.(*ast.CallExpr)
					if ok {
						if name, ok := call.Fun.(*ast.Ident); ok && name.Name == "workerSecrets" {
							found = true
						}
					}
					return true
				})
			}
		}
	}
	if !found {
		t.Fatal("workerResult does not use the shared workerSecrets collector")
	}
}

func pilotContains(values []string, wanted string) bool {
	for _, value := range values {
		if value == wanted {
			return true
		}
	}
	return false
}

func TestPilotWorkerResultPreservesRedactionAndTextBoundsAfterExtraction(t *testing.T) {
	req := harness.Request{APIKeyEnv: "SELECTED_PROVIDER", Env: map[string]string{"SELECTED_PROVIDER": "provider-selected-secret", "EXTRA_TOKEN": "other-selected-secret"}}
	text := "broker-owner-token provider-selected-secret other-selected-secret native-selected-secret ordinary-non-secret 知識知識知識"
	clean := workerResult(text, req, "broker-owner-token", 256, "native-selected-secret")
	for _, value := range workerSecrets(req, "broker-owner-token", "native-selected-secret") {
		if strings.Contains(clean, value) {
			t.Fatalf("workerResult lost selected-secret redaction: %q", clean)
		}
	}
	if !strings.Contains(clean, "ordinary-non-secret") {
		t.Fatal("unrelated text was removed")
	}
	for _, limit := range []int{0, 1, 2, 3, 4, 12, 16, 64} {
		clean := workerResult(text, req, "broker-owner-token", limit, "native-selected-secret")
		if len(clean) > limit || !utf8.ValidString(clean) {
			t.Fatalf("bounded UTF-8 output regressed at %d: %q", limit, clean)
		}
	}
}
