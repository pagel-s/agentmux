	"encoding/json"

	"os"
	"path/filepath"
	"strings"
	"testing"
)

type pilotTraceAppendOutput struct {
	buffer bytes.Buffer
	path   string
	once   bool
}

func (w *pilotTraceAppendOutput) Write(p []byte) (int, error) {
	if !w.once {
		w.once = true
		f, err := os.OpenFile(w.path, os.O_APPEND|os.O_WRONLY, 0600)
		if err != nil {
			return 0, err
		}
		_, err = f.WriteString("UNAUTHENTICATED-CONTENT-AFTER-VERIFICATION\n")
		f.Close()
		if err != nil {
			return 0, err
		}
	}
	return w.buffer.Write(p)
}

func TestPilotAuthenticatedTraceDisplay(t *testing.T) {
	state := filepath.Join(t.TempDir(), "state")
	s, err := managed.Init(state, managedconfig.Config{Version: 2, Mode: "managed", Session: "trace-display", Agents: map[string]managedconfig.Agent{"agent": {Harness: "command", Command: []string{"/usr/bin/true"}}}})
	if err != nil {
		t.Fatal(err)
	}
	s.Close()
	dir := filepath.Join(state, "traces", "agent")
	if err := os.MkdirAll(dir, 0700); err != nil {
		t.Fatal(err)
	}
	key := bytes.Repeat([]byte{0x42}, 32)
	if err := os.WriteFile(filepath.Join(state, "traces", "trace.key"), key, 0600); err != nil {
		t.Fatal(err)
	}
	id := "task_pilot-attempt_display"
	input, _ := json.Marshal(map[string]string{"input": strings.Repeat("verified input ", 5000)})
	records := []managed.TraceRecord{
		{Agent: "agent", TaskID: "task_pilot", TraceID: id, Sequence: 1, At: "2026-11-03T00:00:00Z", Kind: "input", Data: input},
		{Agent: "agent", TaskID: "task_pilot", TraceID: id, Sequence: 2, At: "2026-11-03T00:00:01Z", Kind: "end", Data: json.RawMessage(`{"status":"submitted"}`)},
	}
	var original bytes.Buffer
	previous := ""
	for _, record := range records {
		record.Previous = previous
		unsigned, _ := json.Marshal(record)
		mac := hmac.New(sha256.New, key)
		mac.Write(unsigned)
		record.MAC = hex.EncodeToString(mac.Sum(nil))
		previous = record.MAC
		encoded, _ := json.Marshal(record)
		original.Write(encoded)
		original.WriteByte('\n')
	}
	path := filepath.Join(dir, id+".jsonl")
	if err := os.WriteFile(path, original.Bytes(), 0600); err != nil {
		t.Fatal(err)
	}
	if _, err := managed.VerifyTrace(state, "agent", id, previous); err != nil {
		t.Fatalf("fixture is not authenticated: %v", err)
	}
	out := &pilotTraceAppendOutput{path: path}
	var stderr bytes.Buffer
	if code := runManagedTrace(context.Background(), []string{"show", "--state", state, "--agent", "agent", "--id", id, "--mac", previous}, out, &stderr); code != 0 || !bytes.Equal(out.buffer.Bytes(), original.Bytes()) {
		t.Fatalf("show published a changed source: code=%d bytes=%d stderr=%s", code, out.buffer.Len(), stderr.String())
	}
	if _, err := managed.VerifyTrace(state, "agent", id, previous); err == nil {
		t.Fatal("fixture did not append unauthenticated source content")
	}
	var refused bytes.Buffer
	if code := runManagedTrace(context.Background(), []string{"show", "--state", state, "--agent", "agent", "--id", id, "--mac", previous}, &refused, &stderr); code == 0 || bytes.Contains(refused.Bytes(), []byte(strings.Repeat("verified input ", 3))) {
		t.Fatalf("authentication failure published trace bytes: code=%d bytes=%d", code, refused.Len())
	}
	if err := os.WriteFile(path, original.Bytes(), 0600); err != nil {
		t.Fatal(err)
	}
	if code := runManagedTrace(context.Background(), []string{"show", "--state", state, "--agent", "agent", "--id", id, "--mac", previous}, pilotTraceFailingOutput{}, &stderr); code == 0 {
		t.Fatal("output failure reported success")
	}
}

type pilotTraceFailingOutput struct{}

func (pilotTraceFailingOutput) Write([]byte) (int, error) { return 0, io.ErrClosedPipe }
